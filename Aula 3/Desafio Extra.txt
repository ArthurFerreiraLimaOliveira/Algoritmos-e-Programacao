4-Java

import java.util.Scanner;

public class AluguelBicicletas {
public static void main(String[] args) {

Scanner leia = new Scanner(System.in);

final double VALOR_HORA = 8.00;
String Cliente;
int horas;
double valorTotal;

System.out.print("Digite o nome do cliente: ");
Cliente = leia.nextLine();

System.out.print("Digite a quantidade de horas utilizadas: ");
horas = leia.nextInt();

valorTotal = horas * VALOR_HORA;

System.out.println("Cliente: " + Cliente);
System.out.println("Horas utilizadas: " + horas);
System.out.printf("Valor total: R$ %.2f\n", valorTotal);

if (valorTotal > 50) {
System.out.println("Categoria: Cliente Premium");
}
else {
System.out.println("Categoria: Cliente Comum");
}
}
}
